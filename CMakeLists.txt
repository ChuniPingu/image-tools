cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/build-manifest.json" IMAGE_TOOLS_MANIFEST)
string(JSON IMAGE_TOOLS_VERSION GET "${IMAGE_TOOLS_MANIFEST}" version)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(image-tools VERSION "${IMAGE_TOOLS_VERSION}" LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "image-tools currently builds for Windows x64 with an MSVC-compatible compiler.")
endif()
if(DEFINED MSVC_CXX_ARCHITECTURE_ID AND NOT MSVC_CXX_ARCHITECTURE_ID STREQUAL "x64")
    message(FATAL_ERROR "Configure the Windows x64 target (-A x64).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(cmake/VipsSdk.cmake)
include(CTest)

add_library(image_tools_raster STATIC src/raster_pipeline.cpp)
target_include_directories(image_tools_raster PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(image_tools_raster PUBLIC Vips::Vips)
target_compile_options(image_tools_raster PRIVATE /W4 /utf-8)
target_compile_definitions(image_tools_raster PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(image-tools src/main.cpp)
target_link_libraries(image-tools PRIVATE image_tools_raster)
target_compile_options(image-tools PRIVATE /W4 /utf-8)
target_compile_definitions(image-tools PRIVATE IMAGE_TOOLS_VERSION="${PROJECT_VERSION}")

function(image_tools_copy_runtime target)
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${VIPS_RUNTIME_DLL}" "$<TARGET_FILE_DIR:${target}>"
        VERBATIM)
endfunction()
image_tools_copy_runtime(image-tools)
if(BUILD_TESTING)
    add_executable(raster_tests tests/raster_tests.cpp)
    target_link_libraries(raster_tests PRIVATE image_tools_raster)
    target_compile_options(raster_tests PRIVATE /W4 /utf-8)
    image_tools_copy_runtime(raster_tests)
    add_test(NAME raster_tests COMMAND raster_tests)
    add_test(NAME cli_version COMMAND image-tools --version)
    set_tests_properties(cli_version PROPERTIES PASS_REGULAR_EXPRESSION "image-tools ${PROJECT_VERSION}")
endif()

find_package(Git QUIET)
set(IMAGE_TOOLS_COMMIT "unknown")
if(Git_FOUND)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        OUTPUT_VARIABLE _git_commit OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET RESULT_VARIABLE _git_result)
    if(_git_result EQUAL 0)
        set(IMAGE_TOOLS_COMMIT "${_git_commit}")
    endif()
endif()
configure_file(cmake/build-metadata.json.in "${CMAKE_BINARY_DIR}/build-metadata.json" @ONLY)
install(TARGETS image-tools RUNTIME DESTINATION .)
install(FILES "${VIPS_RUNTIME_DLL}" DESTINATION .)
install(FILES LICENSE THIRD-PARTY-NOTICES.md build-manifest.json DESTINATION .)
install(FILES "${VIPS_SDK_ROOT}/LICENSE" DESTINATION licenses RENAME libvips-LICENSE.txt)
install(FILES licenses/GPL-3.0.txt licenses/LGPL-3.0.txt
    licenses/libvips-THIRD-PARTY-NOTICES.md DESTINATION licenses)
install(DIRECTORY licenses/dependencies DESTINATION licenses)
install(FILES "${VIPS_SDK_ROOT}/versions.json" DESTINATION . RENAME dependency-versions.json)
install(FILES "${CMAKE_BINARY_DIR}/build-metadata.json" DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME image-tools-win-x64)
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/artifacts")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
include(CPack)
